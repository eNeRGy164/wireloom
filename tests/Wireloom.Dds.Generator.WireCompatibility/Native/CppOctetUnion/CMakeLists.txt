cmake_minimum_required(VERSION 3.20)
project(WireloomCppOctetUnion LANGUAGES CXX)
list(APPEND CMAKE_MODULE_PATH "$ENV{NDDSHOME}/resource/cmake")
find_package(RTIConnextDDS 7.7.0 EXACT REQUIRED)
get_filename_component(repository "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(oracle "${repository}/docs/corpus/7.7.0/rtiddsgen/cpp/features/07-union-octet")
set(WIRELOOM_CPP_GENERATED_DIR "${repository}/docs/corpus/7.7.0/generator/cpp/features/07-union-octet" CACHE PATH "Directory containing retained Wireloom output")
function(add_peer target source)
    add_executable(${target} peer.cpp "${source}/07-union-octet.cxx" "${source}/07-union-octetPlugin.cxx")
    target_include_directories(${target} PRIVATE "${source}")
    target_link_libraries(${target} PRIVATE RTIConnextDDS::cpp2_api)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
endfunction()
add_peer(rti-cpp-octet-union "${oracle}")
add_peer(wireloom-cpp-octet-union "${WIRELOOM_CPP_GENERATED_DIR}")
enable_testing()
foreach(peer IN ITEMS rti wireloom)
    add_test(NAME ${peer}-octet-union-api COMMAND ${peer}-cpp-octet-union --api)
    add_test(NAME ${peer}-octet-union-dynamic-type COMMAND ${peer}-cpp-octet-union --dynamic-type)
    add_test(NAME ${peer}-octet-union-registration COMMAND ${peer}-cpp-octet-union --registration)
endforeach()
foreach(state IN ITEMS defaults minimum maximum fallback)
    add_test(NAME rti-octet-union-cdr-${state} COMMAND rti-cpp-octet-union --cdr "${CMAKE_CURRENT_BINARY_DIR}/rti-${state}.cdr" ${state})
    add_test(NAME wireloom-octet-union-cdr-${state} COMMAND wireloom-cpp-octet-union --cdr "${CMAKE_CURRENT_BINARY_DIR}/wireloom-${state}.cdr" ${state})
    add_test(NAME octet-union-cdr-equality-${state} COMMAND "${CMAKE_COMMAND}" -E compare_files "${CMAKE_CURRENT_BINARY_DIR}/rti-${state}.cdr" "${CMAKE_CURRENT_BINARY_DIR}/wireloom-${state}.cdr")
    set_tests_properties(octet-union-cdr-equality-${state} PROPERTIES DEPENDS "rti-octet-union-cdr-${state};wireloom-octet-union-cdr-${state}")
endforeach()
set_tests_properties(rti-octet-union-api rti-octet-union-dynamic-type rti-octet-union-registration wireloom-octet-union-api wireloom-octet-union-dynamic-type wireloom-octet-union-registration PROPERTIES TIMEOUT 15)
foreach(state IN ITEMS defaults minimum maximum fallback)
    set_tests_properties(rti-octet-union-cdr-${state} wireloom-octet-union-cdr-${state} octet-union-cdr-equality-${state} PROPERTIES TIMEOUT 15)
endforeach()
